cmake_minimum_required(VERSION 3.20)
project(SerialSpark VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CLI11 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
add_library(serialspark_core src/core.cpp src/serial.cpp src/baud.cpp)
target_include_directories(serialspark_core PUBLIC include)
target_link_libraries(serialspark_core PUBLIC nlohmann_json::nlohmann_json PkgConfig::UDEV)
target_compile_options(serialspark_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(serialspark src/main.cpp)
target_link_libraries(serialspark PRIVATE serialspark_core CLI11::CLI11)
target_compile_options(serialspark PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_test tests/core_test.cpp)
  target_link_libraries(core_test PRIVATE serialspark_core)
  add_test(NAME core COMMAND core_test)
  add_executable(serial_test tests/serial_test.cpp)
  target_link_libraries(serial_test PRIVATE serialspark_core util)
  add_test(NAME serial COMMAND serial_test)
endif()
install(TARGETS serialspark RUNTIME DESTINATION bin)
set(CPACK_PACKAGE_CONTACT "SerialSpark maintainers")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Serial communication and protocol debugging CLI")
set(CPACK_GENERATOR DEB)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
